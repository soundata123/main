data class UploadJob(
    val id: String,
    val text: String,
    val fileUri: Uri?,
    val completion: CompletableDeferred<Boolean> = CompletableDeferred()
)

data class UploadResponse(val status: String, val id: String)
data class DataEntryResponse(val id: String, val text: String, val fileUrl: String?, val fileType: String?)

interface ApiService {
    @Multipart
    @POST("protocol/upload")
    suspend fun uploadData(
        @Part("text") text: RequestBody,
        @Part("id") id: RequestBody,
        @Part file: MultipartBody.Part?
    ): UploadResponse

    @GET("protocol/get")
    suspend fun getById(
        @Query("id") id: String
    ): DataEntryResponse

    @Streaming
    @GET("protocol/download")
    fun downloadFile(
        @Query("id") id: String
    ): Call<okhttp3.ResponseBody>
}

    private fun getFileSize(uri: Uri): Long {
        contentResolver.query(uri, null, null, null, null)?.use { cursor ->
            val sizeIndex = cursor.getColumnIndex(OpenableColumns.SIZE)
            if (sizeIndex != -1) {
                cursor.moveToFirst()
                return cursor.getLong(sizeIndex)
            }
        }
        return -1
    }

    private suspend fun processUploadQueue() {
        for (job in uploadQueue) {
            var attempts = 0
            val maxRetries = 3
            var success = false

            while (attempts < maxRetries && !success) {
                try {
                    uploadData(job.id, job.text, job.fileUri)
                    success = true
                } catch (e: Exception) {
                    attempts++
                    delay(2000L)
                }
            }

            if (!success) {
                withContext(Dispatchers.Main) {
                    _uploadStatus.value = "fallito dopo $maxRetries tentativi"
                }
            }

            job.completion.complete(success)
        }
    }

    private suspend fun uploadData(id: String, text: String, fileUri: Uri?) {
        val textBody = text.toRequestBody("text/plain".toMediaTypeOrNull())
        val idBody = id.toRequestBody("text/plain".toMediaTypeOrNull())

        val filePart = fileUri?.let { uri ->
            val file = copyUriToFile(uri)
            val mimeType = contentResolver.getType(uri) ?: "*/*"
            val requestFile = file.asRequestBody(mimeType.toMediaTypeOrNull())
            MultipartBody.Part.createFormData("file", file.name, requestFile)
        }

        val response = apiService.uploadData(textBody, idBody, filePart)

        withContext(Dispatchers.Main) {
            _uploadStatus.value = "riuscito con id: ${response.id}"
            showNotification("Dati inviati", "ID: ${response.id}")
        }
    }

    private fun getFileNameFromUri(uri: Uri): String? {
        return when (uri.scheme) {
            "content" -> contentResolver.query(uri, null, null, null, null)?.use { cursor ->
                if (cursor.moveToFirst()) cursor.getString(cursor.getColumnIndexOrThrow(OpenableColumns.DISPLAY_NAME))
                else null
            }
            "file" -> uri.lastPathSegment
            else -> null
        }
    }

    private suspend fun downloadFile(id: String, destinationUri: Uri) {
        try {
            val response = apiService.downloadFile(id).execute()
            if (response.isSuccessful) {
                response.body()?.let { body ->
                    contentResolver.openOutputStream(destinationUri)?.use { outputStream ->
                        body.byteStream().use { inputStream ->
                            inputStream.copyTo(outputStream)
                            _uploadStatus.value = "File scaricato con successo"
                            showNotification("Download completato", "File salvato")
                        }
                    }
                } ?: run {
                    _uploadStatus.value = "Nessun contenuto nel file"
                }
            } else {
                _uploadStatus.value = "Errore durante il download: ${response.message()}"
            }
        } catch (e: Exception) {
            _uploadStatus.value = "Errore durante il download: ${e.message}"
        }
    }

    private suspend fun playIdSequence(id: String) = withContext(Dispatchers.IO) {
        val durationMs = 200
        val toneFrequencies = id.mapNotNull { charToFreq[it] }
        val samplesPerTone = (sampleRate * durationMs / 1000.0).toInt()
        val bufferSize = AudioTrack.getMinBufferSize(
            sampleRate,
            AudioFormat.CHANNEL_OUT_MONO,
            AudioFormat.ENCODING_PCM_16BIT
        )

        val audioTrack = AudioTrack(
            AudioAttributes.Builder()
                .setUsage(AudioAttributes.USAGE_MEDIA)
                .setContentType(AudioAttributes.CONTENT_TYPE_MUSIC)
                .build(),
            AudioFormat.Builder()
                .setEncoding(AudioFormat.ENCODING_PCM_16BIT)
                .setSampleRate(sampleRate)
                .setChannelMask(AudioFormat.CHANNEL_OUT_MONO)
                .build(),
            bufferSize,
            AudioTrack.MODE_STREAM,
            AudioManager.AUDIO_SESSION_ID_GENERATE
        )

        audioTrack.play()

        for (freq in toneFrequencies) {
            val buffer = ByteArray(samplesPerTone * 2)
            for (i in 0 until samplesPerTone) {
                val sample = sin(2 * PI * freq * i / sampleRate)
                val volume = 0.5  // 70%
                val value = (sample * Short.MAX_VALUE * volume).toInt()

                buffer[i * 2] = (value and 0xFF).toByte()
                buffer[i * 2 + 1] = ((value shr 8) and 0xFF).toByte()
            }
            audioTrack.write(buffer, 0, buffer.size)
        }

        audioTrack.stop()
        audioTrack.release()
    }

    private fun startListening() {
        if (checkSelfPermission(Manifest.permission.RECORD_AUDIO) != PackageManager.PERMISSION_GRANTED) {
            _uploadStatus.value = "Permesso microfono non concesso"
            return
        }

        val validFrequencies = charToFreq.values
        val minValidFreq = validFrequencies.minOrNull() ?: 840.0
        val maxValidFreq = validFrequencies.maxOrNull() ?: 4460.0
        val toleranceHz = 50.0

        fun findClosestCharWithTolerance(freq: Double): Char? {
            return charToFreq.entries.minByOrNull { abs(it.value - freq) }?.takeIf {
                abs(it.value - freq) <= toleranceHz
            }?.key
        }

        listeningJob?.cancel()
        listeningJob = coroutineScope.launch(Dispatchers.Default) {
            val minBufferSize = AudioRecord.getMinBufferSize(
                sampleRate,
                AudioFormat.CHANNEL_IN_MONO,
                AudioFormat.ENCODING_PCM_16BIT
            )

            val audioRecord = AudioRecord(
                MediaRecorder.AudioSource.MIC,
                sampleRate,
                AudioFormat.CHANNEL_IN_MONO,
                AudioFormat.ENCODING_PCM_16BIT,
                minBufferSize
            )

            val buffer = ShortArray(minBufferSize)
            audioRecord.startRecording()

            val detectedChars = mutableListOf<Char>()
            var lastDetectionTime = System.currentTimeMillis()

            try {
                while (isActive) {
                    val read = audioRecord.read(buffer, 0, minBufferSize)
                    if (read > 0) {
                        val peakFreq = detectPeakFrequency(buffer, read)

                        if (peakFreq != null && peakFreq in minValidFreq..maxValidFreq) {
                            val matchedChar = findClosestCharWithTolerance(peakFreq)
                            val now = System.currentTimeMillis()

                            if (matchedChar != null) {
                                if (now - lastDetectionTime >= 180) {
                                    detectedChars.add(matchedChar)
                                    lastDetectionTime = now
                                }

                                if (detectedChars.size == 5) {
                                    val detectedId = detectedChars.joinToString("")
                                    _uploadStatus.value = "🔊 ID rilevato: $detectedId"
                                    showNotification("ID Rilevato", "Trovato ID: $detectedId")

                                    try {
                                        val data = apiService.getById(detectedId)
                                        _receivedData.value = data
                                        _uploadStatus.value = "✅ Dati ricevuti con successo"
                                        showNotification("Dati Ricevuti",
                                            "Testo: ${data.text.take(30)}..." +
                                                    (data.fileUrl?.let { "\nContiene file" } ?: ""))
                                    } catch (e: Exception) {
                                        _uploadStatus.value = "❌ Errore recupero dati: ${e.message}"
                                    }

                                    detectedChars.clear()
                                }
                            } else {
                                detectedChars.clear()
                            }
                        } else {
                            detectedChars.clear()
                        }
                    }
                }
            } finally {
                audioRecord.stop()
                audioRecord.release()
            }
        }
    }

    private fun detectPeakFrequency(buffer: ShortArray, length: Int): Double? {
        val n = length
        val fftSize = 1 shl (ceil(log2(n.toDouble())).toInt())
        val windowed = DoubleArray(fftSize) { i ->
            if (i < n) buffer[i].toDouble() * (0.5 * (1 - cos(2 * PI * i / (n - 1))))
            else 0.0
        }

        val fftResult = fft(windowed)
        val magnitudes = fftResult.map { it.abs() }

        return magnitudes.subList(0, fftSize / 2)
            .withIndex()
            .maxByOrNull { it.value }
            ?.let { maxIndex ->
                maxIndex.index * (sampleRate.toDouble() / fftSize)
            }
    }

    private fun fft(input: DoubleArray): Array<Complex> {
        val n = input.size
        if (n == 1) return arrayOf(Complex(input[0], 0.0))

        val even = DoubleArray(n / 2) { input[2 * it] }
        val odd = DoubleArray(n / 2) { input[2 * it + 1] }

        val fftEven = fft(even)
        val fftOdd = fft(odd)

        return Array(n) { k ->
            if (k < n / 2) {
                val twiddle = Complex.polar(1.0, -2 * PI * k / n) * fftOdd[k]
                fftEven[k] + twiddle
            } else {
                val twiddle = Complex.polar(1.0, -2 * PI * (k - n / 2) / n) * fftOdd[k - n / 2]
                fftEven[k - n / 2] - twiddle
            }
        }
    }

    private fun copyUriToFile(uri: Uri): File {
        val inputStream = contentResolver.openInputStream(uri) ?: throw IOException("Impossibile aprire uri")
        val tempFile = File(cacheDir, "upload_${System.currentTimeMillis()}.tmp")
        inputStream.use { input ->
            tempFile.outputStream().use { output ->
                input.copyTo(output)
            }
        }
        return tempFile
    }

    private data class Complex(val re: Double, val im: Double) {
        operator fun plus(other: Complex) = Complex(re + other.re, im + other.im)
        operator fun minus(other: Complex) = Complex(re - other.re, im - other.im)
        operator fun times(other: Complex) = Complex(
            re * other.re - im * other.im,
            re * other.im + im * other.re
        )

        fun abs() = sqrt(re * re + im * im)

        companion object {
            fun polar(r: Double, theta: Double) = Complex(r * cos(theta), r * sin(theta))
        }
    }
}

class AudioForegroundService : Service() {
    override fun onBind(intent: Intent?) = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        val notification = NotificationCompat.Builder(this, "audio_service_channel")
            .setContentTitle("Audio Capture")
            .setContentText("In ascolto per trasmissioni dati...")
            .setSmallIcon(android.R.drawable.ic_btn_speak_now)
            .setPriority(NotificationCompat.PRIORITY_LOW)
            .setOngoing(true)
            .build()

        startForeground(1, notification)
        return START_STICKY
    }
